generarIdFactura(nombre, apellidos, fecha, mesStr) { let maxNum = 0; if (this.pagosList && this.pagosList.length > 0) { const numeros = this.pagosList.map(p => { if (p.factura_id) { const partes = String(p.factura_id).split('-'); const numeroFrente = parseInt(partes[0]); return isNaN(numeroFrente) ? 0 : numeroFrente; } return 0; }); maxNum = Math.max(...numeros, 0); } const siguienteNum = maxNum + 1; // 1. Limpiamos tildes totalmente para que Supabase no machaque archivos let nombreLimpio = String(nombre || '').normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim().toUpperCase(); let apellidoLimpio = String(apellidos || '').normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim().toUpperCase(); const palabras = nombreLimpio.split(/\s+/).filter(w => w.length > 0); const primerNombre = palabras[0] || ''; // 2. Extraemos las 3 primeras letras del nombre let iniciales = "XXX"; if (primerNombre.length >= 3) { iniciales = primerNombre.substring(0, 3); } else if (primerNombre.length > 0) { iniciales = primerNombre.padEnd(3, 'X'); } // 3. Añadimos la 1ª letra del apellido para evitar colisiones (Ej: ERIG vs ERIR) let inicialApellido = apellidoLimpio.length > 0 ? apellidoLimpio.charAt(0) : 'X'; let identificadorUnico = `${iniciales}${inicialApellido}`; const mesLimpio = String(mesStr || '').trim().split(' ')[0]; const mesLetras = (mesLimpio.length >= 3 ? mesLimpio.substring(0, 3) : "GEN").toUpperCase(); let dia = "01"; if (fecha) { const partesFecha = fecha.split('-'); if (partesFecha.length === 3) dia = partesFecha[2]; } return `${siguienteNum}-${identificadorUnico}-${mesLetras}-${dia}`; }
